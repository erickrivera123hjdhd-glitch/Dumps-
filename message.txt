------------------------------------------------------------------ helpers
local function parseValue(s)
    s = (tostring(s):lower():gsub("[,%s%$]", ""))
    local num, suf = s:match("^([%d%.]+)([kmbt]?)$")
    if not num then return nil end
    local mult = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }
    local n = tonumber(num)
    if not n then return nil end
    return n * (mult[suf] or 1)
end
local function fmt(n)
    local u = { { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }
    for _, x in ipairs(u) do
        if n >= x[1] then
            local s = string.format("%.2f", n / x[1])
            s = (s:gsub("%.?0+$", ""))
            return s .. x[2]
        end
    end
    return tostring(math.floor(n))
end
local function minVal() return parseValue(cfg.minValue) or 0 end
local function ago(t)
    local m = math.floor((os.time() - t) / 60)
    if m < 1 then return "just now" end
    if m < 60 then return m .. "m ago" end
    return math.floor(m / 60) .. "h ago"
end

    end
end

local function getGen(name)
    local e = lookup[string.lower(name)]
    if e then return e.Generation or 0 end
    local base = string.match(name, "^(.-)%s*[%[%(]")
    if base and lookup[string.lower(base)] then return lookup[string.lower(base)].Generation or 0 end
    return nil
end

local OWNER_KEYS = { "Owner", "OwnerId", "OwnerUserId", "OwnerName", "UserId", "Player", "PlayerName" }

local function matchesMe(v)
    if v == nil then return false end
    if typeof(v) == "Instance" then return v == player end
    local s = tostring(v):lower()
    if s == "" then return false end
    return s == tostring(player.UserId) or s == player.Name:lower() or s == player.DisplayName:lower()
end

local function plotOwnedByMe(plot)
    -- 1) attributes on the plot
    for _, key in ipairs(OWNER_KEYS) do
        local ok, v = pcall(function() return plot:GetAttribute(key) end)
        if ok and matchesMe(v) then return true end
    end
    -- 2) value objects on the plot
    for _, key in ipairs(OWNER_KEYS) do
        local o = plot:FindFirstChild(key)
        if o and (o:IsA("ObjectValue") or o:IsA("StringValue") or o:IsA("IntValue") or o:IsA("NumberValue")) then
            if matchesMe(o.Value) then return true end
        end
    end
    -- 3) plot sign (YourBase marker or "<name>'s Base" text)
    local sign = plot:FindFirstChild("PlotSign")
    if sign then
        local yb = sign:FindFirstChild("YourBase", true)
        if yb then
            local ok, en = pcall(function() return yb.Enabled end)
            if ok and en == true then return true end
            local ok2, vis = pcall(function() return yb.Visible end)
            if ok2 and vis == true then return true end
        end
        local n, dn = player.Name:lower(), player.DisplayName:lower()
        for _, d in ipairs(sign:GetDescendants()) do
            if d:IsA("TextLabel") then
                local t = d.Text:lower()
                if t ~= "" and (t:find(n .. "'s", 1, true) or t:find(dn .. "'s", 1, true)
                    or t:find(n .. "’s", 1, true) or t:find(dn .. "’s", 1, true)) then
                    return true
                end
            end
        end
    end
    return false
end

local function plotPos(plot)
    local ok, cf = pcall(function() return plot:GetPivot() end)
    if ok and cf then return cf.Position end
    local p = plot:FindFirstChildWhichIsA("BasePart", true)
    return p and p.Position
end

local function nearestPlot(plots)
    plots = plots or Workspace:FindFirstChild("Plots")
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not (plots and hrp) then return nil end
    local best, bd
    for _, plot in ipairs(plots:GetChildren()) do
        local pos = plotPos(plot)
        if pos then
            local d = (pos - hrp.Position).Magnitude
            if not bd or d < bd then best, bd = plot, d end
        end
    end
    return best
end

local function detectBase(plots)
    if S.manualPlot and S.manualPlot.Parent == plots then
        S.myPlot, S.baseWhy = S.manualPlot, "manual"
        return
    end
    for _, plot in ipairs(plots:GetChildren()) do
        if plotOwnedByMe(plot) then
            S.myPlot, S.baseWhy = plot, "detected"
            return
        end
    end
    if S.guessPlot and S.guessPlot.Parent == plots then
        S.myPlot, S.baseWhy = S.guessPlot, "guess (nearest plot)"
        return
    end
    S.myPlot, S.baseWhy = nil, "not found"
    local g = nearestPlot(plots)
    if g then
        S.guessPlot = g
        S.myPlot, S.baseWhy = g, "guess (nearest plot)"
    end
end

local scanning = false
local GEN_PATTERN = "([%d,%.]+)%s*([KkMmBbTt]?)%s*/%s*[sS]"

local function textGen(txt)
    if type(txt) ~= "string" or txt == "" then return nil end
    txt = (txt:gsub("<[^>]+>", ""))
    local num, suf = txt:match(GEN_PATTERN)
    if not num then return nil end
    num = num:gsub(",", "")
    return parseValue(num .. suf)
end

local function plotOf(inst, plots)
    if not plots then return nil end
    while inst and inst.Parent do
        if inst.Parent == plots then return inst end
        inst = inst.Parent
    end
end

local function ownerOf(lb, plots)
    local inst, fallback = lb, nil
    while inst and inst ~= Workspace do
        if inst:IsA("Attachment") then fallback = inst.Parent end
        if inst:IsA("BasePart") and not fallback then fallback = inst end
        if plots and inst.Parent == plots then break end
        if inst:IsA("Model") then return inst end
        inst = inst.Parent
    end
    return fallback
end

local function scanBrainrots()
    if scanning then return end
    scanning = true
    local ok, err = pcall(function()
        local plots = Workspace:FindFirstChild("Plots")
        if plots then
            detectBase(plots)
        else
            -- bases only: nothing to scan without the Plots folder
            S.myPlot, S.baseWhy = nil, "no Plots folder"
            S.ranked, S.gensMap, S.namesMap = {}, {}, {}
            S.top, S.count, S.ownCount = {}, 0, 0
            S.bestModel, S.bestGen, S.bestName = nil, 0, nil
            S.scanned = true
            for m in pairs(tags) do untag(m) end
            setBrainStatus()
            return
        end

        local found, names, owner = {}, {}, {}

        -- 1) read the "$X/s" text the game shows (includes mutations), bases only
        local n = 0
        for _, d in ipairs(plots:GetDescendants()) do
            n = n + 1
            if n % 4000 == 0 then task.wait() end
            if d:IsA("TextLabel") and d.Visible and d.Text ~= "" and d.Parent then
                local g = textGen(d.Text)
                if g and g > 0 then
                    local bg = d:FindFirstAncestorWhichIsA("BillboardGui")
                    if not (bg and (string.sub(bg.Name, 1, 4) == "BRNT" or bg.Enabled == false)) then
                        local key = ownerOf(d, plots)
                        if key and (not found[key] or g > found[key]) then
                            found[key] = g
                            owner[key] = plotOf(key, plots)
                            local dn = d.Parent:FindFirstChild("DisplayName")
                            names[key] = (dn and dn:IsA("TextLabel") and dn.Text ~= "" and dn.Text) or key.Name
                        end
                    end
                end
            end
        end

        -- 2) fallback: base value from the game's data table (skips ones already found)
        local positions = {}
        for key in pairs(found) do
            local part = key.Parent and findRoot(key)
            if part then positions[#positions + 1] = part.Position end
        end
        for _, plot in ipairs(plots:GetChildren()) do
            for _, model in ipairs(plot:GetChildren()) do
                if model:IsA("Model") and not found[model] then
                    local g = getGen(model.Name)
                    if g and g > 0 then
                        local part = findRoot(model)
                        local dup = false
                        if part then
                            for _, p in ipairs(positions) do
                                if (p - part.Position).Magnitude < 8 then dup = true break end
                            end
                        end
                        if not dup then
                            found[model], names[model], owner[model] = g, model.Name, plot
                        end
                    end
                end
            end
        end

        -- build list
        local list, gens, ownCount = {}, {}, 0
        for key, g in pairs(found) do
            if key.Parent then
                local mine = owner[key] ~= nil and owner[key] == S.myPlot
                if mine then ownCount = ownCount + 1 end
                if not (mine and cfg.hideOwn) then
                    list[#list + 1] = key
                    gens[key] = g
                end
            end
        end
        table.sort(list, function(a, b) return gens[a] > gens[b] end)
        local best = list[1]

        -- keep the ranked list so the ESP can fall back instantly when the best one leaves
        S.ranked, S.gensMap, S.namesMap = list, gens, names

        S.top = {}
        for i = 1, math.min(5, #list) do S.top[i] = { name = names[list[i]], gen = gens[list[i]] } end
        S.count = #list
        S.ownCount = ownCount
        S.bestModel = best
        S.bestGen = best and gens[best] or 0
        S.bestName = best and names[best] or nil
        S.scanned = true

        local desired = {}
        if cfg.brainEsp then
            if best then desired[best] = "best" end
            if cfg.brainAll then
                local mv = minVal()
                for _, m in ipairs(list) do
                    if m ~= best and gens[m] >= mv then desired[m] = "min" end
                end
            end
        end
        for m in pairs(tags) do
            if desired[m] == nil or not m.Parent then untag(m) end
        end
        for m, kind in pairs(desired) do tag(m, gens[m], kind) end

        updateBaseTag()

        local e = findVisited(game.JobId)
        if e and S.bestGen > 0 and e.best ~= S.bestGen then
            e.best = S.bestGen
            saveVisited()
            if refreshSaved then refreshSaved() end
        end
        setBrainStatus()
    end)
    scanning = false
    if not ok then warn("[BrainFind] scan error: " .. tostring(err)) end
end

-- fallback: if the best brainrot leaves its base, jump to the next best one right away
local function stillInBase(m)
    local plots = Workspace:FindFirstChild("Plots")
    return m ~= nil and m.Parent ~= nil and plots ~= nil and m:IsDescendantOf(plots)
end

local function promoteBest()
    local old = S.bestModel
    if not old or stillInBase(old) then return end
    untag(old)
    S.bestModel, S.bestGen, S.bestName = nil, 0, nil
    for _, m in ipairs(S.ranked or {}) do
        if m ~= old and stillInBase(m) then
            S.bestModel = m
            S.bestGen = S.gensMap[m] or 0
            S.bestName = S.namesMap[m]
            if cfg.brainEsp then tag(m, S.bestGen, "best") end
            break
        end
    end
    -- refresh the top list from what is still there
    S.top = {}
    local c = 0
    for _, m in ipairs(S.ranked or {}) do
        if stillInBase(m) then
            c = c + 1
            if c <= 5 then S.top[c] = { name = S.namesMap[m], gen = S.gensMap[m] } end
        end
    end
    S.count = c
    pcall(setBrainStatus)
end

local hopping = false
local lastFail = false
local joinClock = os.clock()

    if p == player then lastFail = true end

local function queueScript()
    local qot = queue_on_teleport or (syn and syn.queue_on_teleport)
    if qot and SCRIPT_URL ~= "" then
        pcall(qot, ('loadstring(game:HttpGet("%s"))()'):format(SCRIPT_URL))
    end
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

local HOSTS = { "games.roblox.com", "games.roproxy.com" }

-- tries every host until one returns a valid server list
local function fetchServers(order, cursor)
    for _, host in ipairs(HOSTS) do
        local url = ("https://%s/v1/games/%d/servers/Public?sortOrder=%s&excludeFullGames=true&limit=100")
            :format(host, game.PlaceId, order)
        if cursor then url = url .. "&cursor=" .. cursor end
        local body = httpGet(url)
        if body then
            local ok, res = pcall(function() return HttpService:JSONDecode(body) end)
            if ok and type(res) == "table" and type(res.data) == "table" then return res end
        end
        task.wait(0.4)
    end
end

-- no server list needed: Roblox picks the server itself
local function randomTeleport()
    lastFail = false
    setHopStatus("Joining a random server...")
    pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
    local t0 = os.clock()
    while alive() and os.clock() - t0 < 10 and not lastFail do task.wait(0.25) end
    return not lastFail
end

local function hop()
    if hopping then return end
    hopping = true
    task.spawn(function()
        queueScript()
        local mode = cfg.hopMode or "Auto"
        local done = false

        if mode ~= "Random only" then
            setHopStatus("Searching for a fresh server...")
            local order = (math.random(2) == 1) and "Asc" or "Desc"
            local cursor, pages = nil, 0
            while alive() and not done and pages < 5 do
                pages = pages + 1
                local res = fetchServers(order, cursor)
                if not res then
                    setHopStatus("Server list failed on all hosts.")
                    break
                end
                local cand = {}
                for _, s in ipairs(res.data) do
                    if s.id ~= game.JobId and s.playing and s.maxPlayers
                        and s.playing < s.maxPlayers - 1 and not isRecent(s.id) then
                        cand[#cand + 1] = s
                    end
                end
                shuffle(cand)
                for _, s in ipairs(cand) do
                    if not alive() then break end
                    markVisited(s.id)
                    lastFail = false
                    setHopStatus(("Joining %s... (%d players)"):format(string.sub(s.id, 1, 8), s.playing))
                    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, player) end)
                    local t0 = os.clock()
                    while alive() and os.clock() - t0 < 10 and not lastFail do task.wait(0.25) end
                    if not lastFail then done = true break end
                end
                cursor = res.nextPageCursor
                if not cursor then break end
                task.wait(0.5)
            end
        end

        if not done and alive() and mode ~= "List only" then
            done = randomTeleport()
        end

        if done then
            task.wait(15)
        else
            setHopStatus("Hop failed. Try again or change Hop Method.")
        end
        hopping = false
    end)
end

markVisited(game.JobId) -- save the server we are in

task.spawn(function()
    local notified = false
    while alive() do
        task.wait(2)
        if cfg.autoHop and not hopping and S.scanned and os.clock() - joinClock >= (tonumber(cfg.scanWait) or 6) then
            if S.bestGen > 0 and S.bestGen >= minVal() then
                if not notified then
                    notified = true
                    setHopStatus("Found $" .. fmt(S.bestGen) .. "/s - staying here.")
                end
            else
                notified = false
                hop()
            end
        end
    end
end)

local function relayout()
    local v = gui.AbsoluteSize
    if v.X <= 0 or v.Y <= 0 then return end
    local sc = baseScale()
    local winW = math.clamp(math.floor(v.X / sc) - 20, MIN_W, MAX_W)
    local winH = math.clamp(math.floor(v.Y / sc) - 20, MIN_H, MAX_H)
    compact = winW < COMPACT_BELOW
    local sw = compact and SIDE_COMPACT or SIDE_FULL

    main.Size = UDim2.fromOffset(winW, winH)
    W.sideBg.Size = UDim2.new(0, sw, 1, 0)
    W.divider.Position = UDim2.new(0, sw, 0, 0)
    W.logo.Size = UDim2.new(0, sw, 0, HEADER_H)
    W.logoName.Visible = not compact
    W.logoVer.Visible = not compact
    W.logoShort.Visible = compact
    W.sidebar.Size = UDim2.new(0, sw, 1, -HEADER_H)
    W.right.Position = UDim2.new(0, sw + 1, 0, 0)
    W.right.Size = UDim2.new(1, -(sw + 1), 1, 0)

    local hideHint = compact or NO_KB
    W.footerR.Visible = not hideHint
    W.footerL.Size = hideHint and UDim2.new(1, -36, 1, 0) or UDim2.new(0.7, -18, 1, 0)

    for _, t in pairs(tabs) do
        t.lbl.Visible = not compact
        t.glyph.Visible = compact
    end
    for _, f in ipairs(layoutHooks) do pcall(f) end

    -- keep the window fully on screen after a resize / rotation
    local hw, hh = winW * sc / 2, winH * sc / 2
    local cx = clampN(main.Position.X.Scale * v.X + main.Position.X.Offset, hw + 6, v.X - hw - 6)
    local cy = clampN(main.Position.Y.Scale * v.Y + main.Position.Y.Offset, hh + 6, v.Y - hh - 6)
    main.Position = UDim2.new(cx / v.X, 0, cy / v.Y, 0)
end

    if open then scale.Scale = baseScale() end

-- slider: applies on release; page scrolling is locked while dragging (touch-friendly)
function UI.slider(page, text, min, max, default, decimals, cb)
    local row = makeRow(page, text, SLIDER_H)
    row.LayoutOrder = nextOrder()
    local valLbl = new("TextLabel", {
        Font = F.med, TextSize = 17, TextColor3 = C.sub, BackgroundTransparency = 1,
        Position = UDim2.new(0.5, 0, 0, 0), Size = UDim2.new(0.5, 0, 0, 40),
        TextXAlignment = Enum.TextXAlignment.Right, Text = "",
    }, row)
    local track = new("Frame", {
        Position = UDim2.new(0, 2, 1, -20), Size = UDim2.new(1, -4, 0, 5), BackgroundColor3 = C.field, BorderSizePixel = 0,
    }, row)
    corner(track, 3)
    local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = C.accent, BorderSizePixel = 0 }, track)
    corner(fill, 3)
    local knob = new("Frame", {
        Size = UDim2.new(0, IS_TOUCH and 20 or 14, 0, IS_TOUCH and 20 or 14), AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0), BackgroundColor3 = C.text, BorderSizePixel = 0, ZIndex = 2,
    }, track)
    corner(knob, 10)
    local hitH = IS_TOUCH and 44 or 28
    local hit = new("TextButton", {
        Size = UDim2.new(1, 0, 0, hitH), Position = UDim2.new(0, 0, 0.5, -hitH / 2), BackgroundTransparency = 1, Text = "",
    }, track)

    local value = math.clamp(tonumber(default) or min, min, max)
    local mult = 10 ^ decimals
    local function render()
        local a = (value - min) / (max - min)
        fill.Size = UDim2.new(a, 0, 1, 0)
        knob.Position = UDim2.new(a, 0, 0.5, 0)
        valLbl.Text = string.format("%." .. decimals .. "f", value)
    end
    render()

    local dragging, activeInput = false, nil
    local function setX(x)
        local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        value = math.floor((min + (max - min) * a) * mult + 0.5) / mult
        render()
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, activeInput = true, i
            page.ScrollingEnabled = false
            setX(i.Position.X)
        end
    end)
    bind(UserInputService.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i == activeInput) then
            setX(i.Position.X)
        end
    end)
    bind(UserInputService.InputEnded, function(i)
        if dragging and (i == activeInput or i.UserInputType == Enum.UserInputType.MouseButton1) then
            dragging = false
            page.ScrollingEnabled = true
            if cb then cb(value) end
        end
    end)
    return { Set = function(v) value = math.clamp(v, min, max) render() end }
end

    local selected = { label = tostring(default), value = (not placeholder) and default or nil }
    local obj = {}
    function obj.setOptions(opts)
        for _, c in ipairs(holder:GetChildren()) do

        end
        if #opts == 0 then
            new("TextLabel", { Text = "(nothing here yet)", Font = F.reg, TextSize = 15, TextColor3 = C.dim,
                BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 36), TextXAlignment = Enum.TextXAlignment.Left }, holder)
        end
        for i, o in ipairs(opts) do
            local lbl = type(o) == "table" and o.label or tostring(o)
            local val = type(o) == "table" and o.value or o
            local b = new("TextButton", {
                Text = lbl, Font = F.reg, TextSize = 16, TextColor3 = C.sub,
                BackgroundColor3 = Color3.new(1, 1, 1), BackgroundTransparency = 1, AutoButtonColor = false,
                Size = UDim2.new(1, 0, 0, IS_TOUCH and 42 or 36), TextXAlignment = Enum.TextXAlignment.Left, LayoutOrder = i,
                TextTruncate = Enum.TextTruncate.AtEnd,

            new("UIPadding", { PaddingLeft = UDim.new(0, 8) }, b)
            b.MouseEnter:Connect(function() tween(b, 0.1, { BackgroundTransparency = 0.94, TextColor3 = C.text }) end)
            b.MouseLeave:Connect(function() tween(b, 0.1, { BackgroundTransparency = 1, TextColor3 = C.sub }) end)
            b.MouseButton1Click:Connect(function()
                selected = { label = lbl, value = val }
                valLbl.Text = lbl
                holder.Visible = false
                if cb then cb(val, lbl) end
            end)
        end
    end
    function obj.get() return selected.value end
    obj.setOptions(options)
    hover(row, rowHit)
    rowHit.MouseButton1Click:Connect(function() holder.Visible = not holder.Visible end)
    return obj
end

local function setOpen(v)
    if v == open then return end
    open = v
    if fab then tween(fab, 0.2, { BackgroundColor3 = fabColor() }) end
    local target = baseScale()
    if v then
        relayout()
        main.Visible = true
        if cfg.anim then
            scale.Scale = target * 0.95
            main.GroupTransparency = 1
            tween(main, 0.18, { GroupTransparency = alphaGoal() })
            tween(scale, 0.18, { Scale = target })
        else
            scale.Scale = target
            main.GroupTransparency = alphaGoal()
        end
    else
        if cfg.anim then
            local tw = tween(main, 0.14, { GroupTransparency = 1 })
            tween(scale, 0.14, { Scale = target * 0.96 })
            tw.Completed:Connect(function() if not open then main.Visible = false end end)
        else
            main.GroupTransparency = 1
            main.Visible = false
        end
    end
end

-- drag + click handler that works with mouse and touch
local function makeDraggable(handle, target, opts)
    opts = opts or {}
    local dragging, moved, activeInput, startPt, startCenter = false, false, nil, nil, nil
    handle.InputBegan:Connect(function(i)
        if dragging then return end
        if opts.canDrag and not opts.canDrag() then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            local v = gui.AbsoluteSize
            dragging, moved, activeInput = true, false, i
            startPt = Vector2.new(i.Position.X, i.Position.Y)
            startCenter = Vector2.new(
                target.Position.X.Scale * v.X + target.Position.X.Offset,
                target.Position.Y.Scale * v.Y + target.Position.Y.Offset)
        end
    end)
    bind(UserInputService.InputChanged, function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i == activeInput then
            local d = Vector2.new(i.Position.X, i.Position.Y) - startPt
            if not moved and d.Magnitude > DRAG_THRESHOLD then moved = true end
            if not moved then return end
            local v = gui.AbsoluteSize
            if v.X <= 0 or v.Y <= 0 then return end
            local hw, hh = target.AbsoluteSize.X / 2, target.AbsoluteSize.Y / 2
            local cx, cy = startCenter.X + d.X, startCenter.Y + d.Y
            if opts.inside then
                cx = clampN(cx, hw, v.X - hw)
                cy = clampN(cy, hh, v.Y - hh)
            else
                -- the window may leave the screen a little, but the header stays reachable
                cx = clampN(cx, 90 - hw, v.X - 90 + hw)
                cy = clampN(cy, hh, v.Y - 50 + hh)
            end
            target.Position = UDim2.new(cx / v.X, 0, cy / v.Y, 0)
        end
    end)
    bind(UserInputService.InputEnded, function(i)
        if not dragging then return end
        local t = activeInput and activeInput.UserInputType
        local same = (i == activeInput)
            or (t == Enum.UserInputType.MouseButton1 and i.UserInputType == Enum.UserInputType.MouseButton1)
            or (t == Enum.UserInputType.Touch and i.UserInputType == Enum.UserInputType.Touch)
        if not same then return end
        dragging = false
        if moved then
            if opts.onEnd then opts.onEnd(target.Position.X.Scale, target.Position.Y.Scale) end
        elseif opts.onClick then
            opts.onClick()
        end
    end)
end
