------------------------------------------------------------------ helpers
local function parseValue(s)
    s = (tostring(s):lower():gsub("[,%s%$]", ""))
    local num, suf = s:match("^([%d%.]+)([kmbt]?)$")
    if not num then return nil end
    local mult = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }
    local n = tonumber(num)
    if not n then return nil end
    return n * (mult[suf] or 1)
end
local function fmt(n)
    local u = { { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }
    for _, x in ipairs(u) do
        if n >= x[1] then
            local s = string.format("%.2f", n / x[1])
            s = (s:gsub("%.?0+$", ""))
            return s .. x[2]
        end
    end
    return tostring(math.floor(n))
end
local function minVal() return parseValue(cfg.minValue) or 0 end
local function ago(t)
    local m = math.floor((os.time() - t) / 60)
    if m < 1 then return "just now" end
    if m < 60 then return m .. "m ago" end
    return math.floor(m / 60) .. "h ago"
end

end

local OWNER_KEYS = { "Owner", "OwnerId", "OwnerUserId", "OwnerName", "UserId", "Player", "PlayerName" }

local function matchesMe(v)
    if v == nil then return false end
    if typeof(v) == "Instance" then return v == player end
    local s = tostring(v):lower()
    if s == "" then return false end
    return s == tostring(player.UserId) or s == player.Name:lower() or s == player.DisplayName:lower()
end

local function plotOwnedByMe(plot)
    for _, key in ipairs(OWNER_KEYS) do
        local ok, v = pcall(function() return plot:GetAttribute(key) end)
        if ok and matchesMe(v) then return true end
    end
    for _, key in ipairs(OWNER_KEYS) do
        local o = plot:FindFirstChild(key)
        if o and (o:IsA("ObjectValue") or o:IsA("StringValue") or o:IsA("IntValue") or o:IsA("NumberValue")) then
            if matchesMe(o.Value) then return true end
        end
    end
    local sign = plot:FindFirstChild("PlotSign")
    if sign then
        local yb = sign:FindFirstChild("YourBase", true)
        if yb then
            local ok, en = pcall(function() return yb.Enabled end)
            if ok and en == true then return true end
            local ok2, vis = pcall(function() return yb.Visible end)
            if ok2 and vis == true then return true end
        end
        local n, dn = player.Name:lower(), player.DisplayName:lower()
        for _, d in ipairs(sign:GetDescendants()) do
            if d:IsA("TextLabel") then
                local t = d.Text:lower()
                if t ~= "" and (t:find(n .. "'s", 1, true) or t:find(dn .. "'s", 1, true)
                    or t:find(n .. "\u{2019}s", 1, true) or t:find(dn .. "\u{2019}s", 1, true)) then
                    return true
                end
            end
        end
    end
    return false
end

local function plotPos(plot)
    local ok, cf = pcall(function() return plot:GetPivot() end)
    if ok and cf then return cf.Position end
    local p = plot:FindFirstChildWhichIsA("BasePart", true)
    return p and p.Position
end

local function nearestPlot(plots)
    plots = plots or Workspace:FindFirstChild("Plots")
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not (plots and hrp) then return nil end
    local best, bd
    for _, plot in ipairs(plots:GetChildren()) do
        local pos = plotPos(plot)
        if pos then
            local d = (pos - hrp.Position).Magnitude
            if not bd or d < bd then best, bd = plot, d end
        end
    end
    return best
end

local function detectBase(plots)
    if S.manualPlot and S.manualPlot.Parent == plots then
        S.myPlot, S.baseWhy = S.manualPlot, "manual"
        return
    end
    for _, plot in ipairs(plots:GetChildren()) do
        if plotOwnedByMe(plot) then
            S.myPlot, S.baseWhy = plot, "detected"
            return
        end
    end
    if S.guessPlot and S.guessPlot.Parent == plots then
        S.myPlot, S.baseWhy = S.guessPlot, "guess (nearest plot)"
        return
    end
    S.myPlot, S.baseWhy = nil, "not found"
    local g = nearestPlot(plots)
    if g then
        S.guessPlot = g
        S.myPlot, S.baseWhy = g, "guess (nearest plot)"
    end
end

        local part = findRoot(m)

                bb.Name = "BRNT_BEST"

local function updateTagText(m)
    local t = tags[m]
    if t and t.bb and t.bb.Parent then
        local lb = t.bb:FindFirstChildOfClass("TextLabel")
        if lb then
            local part = findRoot(m)
            if part and t.bb.Adornee ~= part then t.bb.Adornee = part end
            lb.Text = ((t.kind == "best") and "BEST  $" or "$") .. fmt(t.gen) .. "/s"
            lb.TextColor3 = (t.kind == "best") and GOLD or CYAN
        end
    end
end

        pcall(function() t.bb:Destroy() end)
        tags[m] = nil
    end
end

local scanning = false
local GEN_PATTERN = "([%d,%.]+)%s*([KkMmBbTt]?)%s*/%s*[sS]"

local function textGen(txt)
    if type(txt) ~= "string" or txt == "" then return nil end
    txt = (txt:gsub("<[^>]+>", ""))
    local num, suf = txt:match(GEN_PATTERN)
    if not num then return nil end
    num = num:gsub(",", "")
    return parseValue(num .. (suf or ""))
end

local MATCH_DIST = 16
local CONVEYOR_NAMES = {
    renderedmovinganimals = true, map = true, road = true, shop = true,
}

local function collectAnimals(plots)
    local list = {}
    local function add(m)
        local root = findRoot(m)
        if root then list[#list + 1] = { model = m, pos = root.Position } end
    end
    for _, plot in ipairs(plots:GetChildren()) do
        for _, m in ipairs(plot:GetChildren()) do
            if m:IsA("Model") then add(m) end
        end
    end
    for _, m in ipairs(Workspace:GetChildren()) do
        if m:IsA("Model") and not CONVEYOR_NAMES[string.lower(m.Name)]
            and not Players:GetPlayerFromCharacter(m) then
            add(m)
        end
    end
    return list
end

local function plotOfModelIn(plots, m)
    local inst = m
    while inst and inst.Parent do
        if inst.Parent == plots then return inst end
        inst = inst.Parent
    end
    return nil
end

local function scanBrainrots()
    if scanning then return end
    scanning = true
    local scans = {}

    local okA, errA = pcall(function()
        local plots = Workspace:FindFirstChild("Plots")
        if not plots then
            scans.empty = true
            scans.errA = "no Plots folder"
            return
        end
        detectBase(plots)

        local found, names, owner = {}, {}, {}
        local matched, boards = 0, 0
        local debris = Workspace:FindFirstChild("Debris")
        local candidates = collectAnimals(plots)
        if debris then
            for _, cand in ipairs(candidates) do
                cand.plot = plotOfModelIn(plots, cand.model)
            end
            for _, tpl in ipairs(debris:GetChildren()) do
                pcall(function()
                    if tpl:IsA("BasePart") and tpl.Name == "FastOverheadTemplate" then
                        local oh = tpl:FindFirstChild("AnimalOverhead")
                        local gen = oh and oh:FindFirstChild("Generation")
                        if oh and gen and gen:IsA("TextLabel") and gen.Visible then
                            local g = textGen(gen.Text)
                            if g and g > 0 then
                                boards = boards + 1
                                local tplPos = tpl.Position
                                local best, bd
                                for _, cand in ipairs(candidates) do
                                    if cand.model.Parent then
                                        local d = (cand.pos - tplPos).Magnitude
                                        if d <= MATCH_DIST and (not bd or d < bd) then
                                            best, bd = cand, d
                                        end
                                    end
                                end
                                if best then
                                    matched = matched + 1
                                    local m = best.model
                                    if not found[m] or g > found[m] then
                                        local dn = oh:FindFirstChild("DisplayName")
                                        names[m] = (dn and dn:IsA("TextLabel") and dn.Text ~= "" and dn.Text) or m.Name
                                        found[m] = g
                                        owner[m] = best.plot
                                    end
                                end
                            end
                        end
                    end
                end)
            end
        end
        scans.plots = plots
        scans.found, scans.names, scans.owner = found, names, owner
        scans.boards, scans.matched = boards, matched
    end)

    local okB, errB = pcall(function()
        if scans.empty then
            S.myPlot, S.baseWhy = nil, "no Plots folder"
            S.ranked, S.gensMap, S.namesMap = {}, {}, {}
            S.top, S.count, S.ownCount = {}, 0, 0
            S.bestModel, S.bestGen, S.bestName = nil, 0, nil
            S.scanned = true
            for m in pairs(tags) do untag(m) end
            setBrainStatus()
            return
        end
        local found, names, owner = scans.found or {}, scans.names or {}, scans.owner or {}

        local list, gens, ownCount = {}, {}, 0
        for key, g in pairs(found) do
            if key.Parent then
                local mine = owner[key] ~= nil and owner[key] == S.myPlot
                if mine then ownCount = ownCount + 1 end
                if not (mine and cfg.hideOwn) then
                    list[#list + 1] = key
                    gens[key] = g
                end
            end
        end
        table.sort(list, function(a, b) return gens[a] > gens[b] end)
        local best = list[1]

        S.ranked, S.gensMap, S.namesMap = list, gens, names
        S.top = {}
        for i = 1, math.min(5, #list) do S.top[i] = { name = names[list[i]], gen = gens[list[i]] } end
        S.count = #list
        S.ownCount = ownCount
        S.bestModel = best
        S.bestGen = best and gens[best] or 0
        S.bestName = best and names[best] or nil
        S.scanned = true

        local desired = {}
        if cfg.brainEsp then
            if best then desired[best] = "best" end
            if cfg.brainAll then
                local mv = minVal()
                for _, m in ipairs(list) do
                    if m ~= best and gens[m] >= mv then desired[m] = "min" end
                end
            end
        end
        for m in pairs(tags) do
            if desired[m] == nil or not m.Parent then untag(m) end
        end
        for m, kind in pairs(desired) do
            local t = tags[m]
            if not (t and t.kind == kind and t.gen == gens[m]) then
                tag(m, gens[m], kind)
            end
            rebuildTag(m)
            updateTagText(m)
        end

        updateBaseTag()

        local e = findVisited(game.JobId)
        if e and S.bestGen > 0 and e.best ~= S.bestGen then
            e.best = S.bestGen
            saveVisited()
            if refreshSaved then refreshSaved() end
        end
    end)

    scanning = false
    lastScanInfo.boards = scans.boards or 0
    lastScanInfo.matched = scans.matched or 0
    lastScanInfo.err = (not okA and errA) or (not okB and errB) or nil
    pcall(setBrainStatus)
    if not okA or not okB then
        warn("[BrainFind] scan phase error: " .. tostring((not okA and errA) or errB))
    end
end

local function stillInBase(m)
    local plots = Workspace:FindFirstChild("Plots")
    return m ~= nil and m.Parent ~= nil and plots ~= nil and m:IsDescendantOf(plots)
end

local function promoteBest()
    local old = S.bestModel
    if not old or stillInBase(old) then return end
    untag(old)
    S.bestModel, S.bestGen, S.bestName = nil, 0, nil
    for _, m in ipairs(S.ranked or {}) do
        if m ~= old and stillInBase(m) then
            S.bestModel = m
            S.bestGen = S.gensMap[m] or 0
            S.bestName = S.namesMap[m]
            if cfg.brainEsp then tag(m, S.bestGen, "best") end
            break
        end
    end
    S.top = {}
    local c = 0
    for _, m in ipairs(S.ranked or {}) do
        if stillInBase(m) then
            c = c + 1
            if c <= 5 then S.top[c] = { name = S.namesMap[m], gen = S.gensMap[m] } end
        end
    end
    S.count = c
    pcall(setBrainStatus)
end

local plrObjs = {}
local function destroyPlr(p)
    local o = plrObjs[p]
    if not o then return end
    pcall(function() o.hl:Destroy() end)
    pcall(function() o.bb:Destroy() end)
    killLine(o.line)
    plrObjs[p] = nil
end

-- BIG tracer: black outline + gold core + endpoint dot + halo dot
local brainTracer
local function tracerTh()
    return math.clamp(math.floor(tonumber(cfg.tracerThick) or 6), 1, 16)
end
local function getBrainTracer()
    local th = tracerTh()
    if not brainTracer or brainTracer.th ~= th then
        if brainTracer then
            killLine(brainTracer.outline)
            killLine(brainTracer.main)
            killLine(brainTracer.dot)
            killLine(brainTracer.halo)
        end
        brainTracer = {
            th = th,
            outline = newLine(Color3.new(0, 0, 0)),
            main = newLine(GOLD),
            dot = newDot(GOLD),
            halo = newDot(Color3.new(0, 0, 0)),
        }
        if brainTracer.outline then
            brainTracer.outline.Thickness = th + 4
            brainTracer.outline.Transparency = 0.5
        end
        if brainTracer.main then brainTracer.main.Thickness = th end
        if brainTracer.dot then brainTracer.dot.Radius = th + 4 end
        if brainTracer.halo then
            brainTracer.halo.Radius = th + 9
            brainTracer.halo.Transparency = 0.7
        end
    end
    return brainTracer
end

local function hideBrainTracer()
    if brainTracer then
        if brainTracer.outline then brainTracer.outline.Visible = false end
        if brainTracer.main then brainTracer.main.Visible = false end
        if brainTracer.dot then brainTracer.dot.Visible = false end
        if brainTracer.halo then brainTracer.halo.Visible = false end
    end
end

end

local function queueScript()
    local qot = queue_on_teleport or (syn and syn.queue_on_teleport)
    if qot and SCRIPT_URL ~= "" then
        pcall(qot, ('loadstring(game:HttpGet("%s"))()'):format(SCRIPT_URL))
    end
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

local HOSTS = { "games.roblox.com", "games.roproxy.com" }

local function fetchServers(order, cursor)
    for _, host in ipairs(HOSTS) do
        local url = ("https://%s/v1/games/%d/servers/Public?sortOrder=%s&excludeFullGames=true&limit=100")
            :format(host, game.PlaceId, order)
        if cursor then url = url .. "&cursor=" .. cursor end
        local body = httpGet(url)
        if body then
            local ok, res = pcall(function() return HttpService:JSONDecode(body) end)
            if ok and type(res) == "table" and type(res.data) == "table" then return res end
        end
        task.wait(0.4)
    end
end

local function randomTeleport()
    lastFail = false
    setHopStatus("Joining a random server...")
    pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
    local t0 = os.clock()
    while alive() and os.clock() - t0 < 10 and not lastFail do task.wait(0.25) end
    return not lastFail
end

local function hop()
    if hopping then return end
    hopping = true
    joinClock = os.clock() -- respect Scan Wait after every hop attempt
    task.spawn(function()
        queueScript()
        local mode = cfg.hopMode or "Auto"
        local done = false

        if mode ~= "Random only" then
            setHopStatus("Searching for a fresh server...")
            local order = (math.random(2) == 1) and "Asc" or "Desc"
            local cursor, pages = nil, 0
            while alive() and not done and pages < 5 do
                pages = pages + 1
                local res = fetchServers(order, cursor)
                if not res then
                    setHopStatus("Server list failed on all hosts.")
                    break
                end
                local cand = {}
                for _, s in ipairs(res.data) do
                    if s.id ~= game.JobId and s.playing and s.maxPlayers
                        and s.playing < s.maxPlayers - 1 and not isRecent(s.id) then
                        cand[#cand + 1] = s
                    end
                end
                shuffle(cand)
                for _, s in ipairs(cand) do
                    if not alive() then break end
                    markVisited(s.id)
                    lastFail = false
                    setHopStatus(("Joining %s... (%d players)"):format(string.sub(s.id, 1, 8), s.playing))
                    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, player) end)
                    local t0 = os.clock()
                    while alive() and os.clock() - t0 < 10 and not lastFail do task.wait(0.25) end
                    if not lastFail then done = true break end
                end
                cursor = res.nextPageCursor
                if not cursor then break end
                task.wait(0.5)
            end
        end

        if not done and alive() and mode ~= "List only" then
            done = randomTeleport()
        end

        if done then
            task.wait(15)
        else
            setHopStatus("Hop failed. Try again or change Hop Method.")
        end
        hopping = false
    end)
end

local function joinServer(id)
    if not id or id == "" then return end
    queueScript()
    lastFail = false
    setHopStatus("Joining " .. string.sub(id, 1, 8) .. "...")
    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, id, player) end)
end

markVisited(game.JobId)

task.spawn(function()
    local notified = false
    while alive() do
        task.wait(2)
        if cfg.autoHop and not hopping and S.scanned and os.clock() - joinClock >= (tonumber(cfg.scanWait) or 6) then
            if S.bestGen > 0 and S.bestGen >= minVal() then
                if not notified then
                    notified = true
                    setHopStatus("Found $" .. fmt(S.bestGen) .. "/s - staying here.")
                end
            else
                notified = false
                hop()
            end
        end
    end
end)

    local t = { btn = btn, pill = pill, bar = bar, lbl = lbl, glyph = glyphLbl, page = page,
        title = title, desc = desc, builder = builder, built = false, active = false }
    tabs[name] = t
    table.insert(tabOrder, name)
    btn.MouseEnter:Connect(function()
        if not t.active then tween(lbl, 0.12, { TextColor3 = C.text }) tween(glyphLbl, 0.12, { TextColor3 = C.text }) end
    end)
    btn.MouseLeave:Connect(function()
        if not t.active then tween(lbl, 0.12, { TextColor3 = C.sub }) tween(glyphLbl, 0.12, { TextColor3 = C.sub }) end
    end)
    btn.MouseButton1Click:Connect(function() selectTab(name) end)
    return t
end

-- LAZY: a page's widgets are only created the first time the tab is opened.
selectTab = function(name)
    local t = tabs[name]
    if t and not t.built then
        t.built = true
        if t.builder then
            local ok, err = pcall(t.builder)
            if not ok then warn("[BrainFind] tab build error (" .. tostring(name) .. "): " .. tostring(err)) end
        end
    end
    for n, tt in pairs(tabs) do
        local on = (n == name)
        tt.active = on
        if on and not tt.page.Visible then tt.page.CanvasPosition = Vector2.new(0, 0) end
        tt.page.Visible = on
        tween(tt.pill, 0.18, { BackgroundTransparency = on and 0.88 or 1 })
        tween(tt.bar, 0.15, { BackgroundTransparency = on and 0 or 1 })
        tween(tt.lbl, 0.15, { TextColor3 = on and C.text or C.sub })
        tween(tt.glyph, 0.15, { TextColor3 = on and C.text or C.sub })
        if on then W.headTitle.Text = tt.title W.headDesc.Text = tt.desc end
    end
end

---------------------------------------------------------------- responsive layout
local function relayout()
    local v = gui.AbsoluteSize
    if v.X <= 0 or v.Y <= 0 then return end
    local sc = baseScale()
    local winW = math.clamp(math.floor(v.X / sc) - 20, MIN_W, MAX_W)
    local winH = math.clamp(math.floor(v.Y / sc) - 20, MIN_H, MAX_H)
    compact = winW < COMPACT_BELOW
    local sw = compact and SIDE_COMPACT or SIDE_FULL

    main.Size = UDim2.fromOffset(winW, winH)
    W.sideBg.Size = UDim2.new(0, sw, 1, 0)
    W.divider.Position = UDim2.new(0, sw, 0, 0)
    W.logo.Size = UDim2.new(0, sw, 0, HEADER_H)
    W.logoName.Visible = not compact
    W.logoVer.Visible = not compact
    W.logoShort.Visible = compact
    W.sidebar.Size = UDim2.new(0, sw, 1, -HEADER_H)
    W.right.Position = UDim2.new(0, sw + 1, 0, 0)
    W.right.Size = UDim2.new(1, -(sw + 1), 1, 0)

    if compact then
        W.bestPill.Visible = false
        W.headTitle.Size = UDim2.new(1, -(CLOSE_SIZE + 22), 0, 30)
        W.headDesc.Size = UDim2.new(1, -(CLOSE_SIZE + 22), 0, 20)
    else
        W.bestPill.Visible = true
        W.headTitle.Size = UDim2.new(1, -(CLOSE_SIZE + 160), 0, 30)
        W.headDesc.Size = UDim2.new(1, -(CLOSE_SIZE + 160), 0, 20)
    end

    local hideHint = compact or NO_KB
    W.footerR.Visible = not hideHint
    W.footerL.Size = hideHint and UDim2.new(1, -36, 1, 0) or UDim2.new(0.7, -18, 1, 0)

    for _, t in pairs(tabs) do
        t.lbl.Visible = not compact
        t.glyph.Visible = compact
    end
    for _, f in ipairs(layoutHooks) do pcall(f) end

    local hw, hh = winW * sc / 2, winH * sc / 2
    local cx = clampN(main.Position.X.Scale * v.X + main.Position.X.Offset, hw + 6, v.X - hw - 6)
    local cy = clampN(main.Position.Y.Scale * v.Y + main.Position.Y.Offset, hh + 6, v.Y - hh - 6)
    main.Position = UDim2.new(cx / v.X, 0, cy / v.Y, 0)
end
gui:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
    relayout()
    if open then scale.Scale = baseScale() end
end)
relayout()

local function hover(row, hit)

        tween(row, 0.1, { BackgroundColor3 = C.card })

    end)
end

function UI.slider(page, text, min, max, default, decimals, cb)
    local row = makeRow(page, text, SLIDER_H)
    row.LayoutOrder = nextOrder()
    local valLbl = new("TextLabel", {
        Font = F.med, TextSize = 18, TextColor3 = C.text, BackgroundTransparency = 1,
        Position = UDim2.new(0.5, 0, 0, 0), Size = UDim2.new(0.5, 0, 0, 40),

    }, row)

        Position = UDim2.new(0, 2, 1, -20), Size = UDim2.new(1, -4, 0, 6), BackgroundColor3 = C.field, BorderSizePixel = 0,
    }, row)
    corner(track, 3)
    local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0 }, track)
    corner(fill, 3)
    new("UIGradient", { Color = ColorSequence.new(C.accent, C.accentSoft) }, fill)
    local knob = new("Frame", {
        Size = UDim2.new(0, IS_TOUCH and 22 or 16, 0, IS_TOUCH and 22 or 16), AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0), BackgroundColor3 = C.text, BorderSizePixel = 0, ZIndex = 2,
    }, track)
    corner(knob, 11)
    local hitH = IS_TOUCH and 44 or 28
    local hit = new("TextButton", {
        Size = UDim2.new(1, 0, 0, hitH), Position = UDim2.new(0, 0, 0.5, -hitH / 2), BackgroundTransparency = 1, Text = "",
    }, track)

    local value = math.clamp(tonumber(default) or min, min, max)
    local mult = 10 ^ decimals
    local function render()
        local a = (value - min) / (max - min)
        fill.Size = UDim2.new(a, 0, 1, 0)
        knob.Position = UDim2.new(a, 0, 0.5, 0)
        valLbl.Text = string.format("%." .. decimals .. "f", value)
    end
    render()

    local dragging, activeInput = false, nil
    local function setX(x)
        local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        value = math.floor((min + (max - min) * a) * mult + 0.5) / mult
        render()
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, activeInput = true, i
            page.ScrollingEnabled = false
            setX(i.Position.X)
        end
    end)
    bind(UserInputService.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i == activeInput) then
            setX(i.Position.X)
        end
    end)
    bind(UserInputService.InputEnded, function(i)
        if dragging and (i == activeInput or i.UserInputType == Enum.UserInputType.MouseButton1) then
            dragging = false
            page.ScrollingEnabled = true
            if cb then cb(value) end
        end
    end)
    return { Set = function(v) value = math.clamp(v, min, max) render() end }
end

---------------------------------------------------------------- open / close
local fab, fabStroke
local function fabColor() return open and FAB_OPEN or FAB_IDLE end

local function setOpen(v)
    if v == open then return end
    open = v
    if fab then tween(fab, 0.2, { BackgroundColor3 = fabColor() }) end
    local target = baseScale()
    if v then
        relayout()
        main.Visible = true
        if cfg.anim then
            scale.Scale = target * 0.95
            main.GroupTransparency = 1
            tween(main, 0.18, { GroupTransparency = alphaGoal() })
            tween(scale, 0.18, { Scale = target }, Enum.EasingStyle.Back)
        else
            scale.Scale = target
            main.GroupTransparency = alphaGoal()
        end
    else
        if cfg.anim then
            local tw = tween(main, 0.14, { GroupTransparency = 1 })
            tween(scale, 0.14, { Scale = target * 0.96 })
            tw.Completed:Connect(function() if not open then main.Visible = false end end)
        else
            main.GroupTransparency = 1
            main.Visible = false
        end
    end
end
W.closeBtn.MouseButton1Click:Connect(function() setOpen(false) end)

---------------------------------------------------------------- unload
local function unload()
    genv.BRNT_TOKEN = (genv.BRNT_TOKEN or 0) + 1 -- kills every loop
    for m in pairs(tags) do untag(m) end
    for p in pairs(plrObjs) do destroyPlr(p) end
    if baseTag then pcall(function() baseTag.bb:Destroy() end) baseTag = nil end
    hideBrainTracer()
    for _, l in ipairs(allLines) do killLine(l) end
    for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    gui:Destroy()
end

        end
    end
    W.footerL.Text = string.format("Server %s   |   %d found", string.sub(game.JobId, 1, 8), S.count or 0)

    if cfg.notify and S.bestModel and good and S.bestModel ~= lastAnnounced then
        lastAnnounced = S.bestModel

    end
end

local function applyMin(txt, box)

local function makeDrag(handle, target, opts)
    local dragging, startInput, startPt, startCenter, moved = false, nil, nil, nil, false
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if opts.canDrag and not opts.canDrag() and not opts.onClick then return end
            local v = gui.AbsoluteSize
            dragging, startInput, moved = true, input, false
            startPt = Vector2.new(input.Position.X, input.Position.Y)
            startCenter = Vector2.new(
                target.Position.X.Scale * v.X + target.Position.X.Offset,
                target.Position.Y.Scale * v.Y + target.Position.Y.Offset)
        end
    end)
    bind(UserInputService.InputChanged, function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input == startInput then
            if opts.canDrag and not opts.canDrag() then return end
            local delta = Vector2.new(input.Position.X, input.Position.Y) - startPt
            if not moved and delta.Magnitude > DRAG_THRESHOLD then moved = true end
            if moved then
                local v = gui.AbsoluteSize
                local half = target.AbsoluteSize / 2
                local cx = clampN(startCenter.X + delta.X, half.X + 4, v.X - half.X - 4)
                local cy = clampN(startCenter.Y + delta.Y, half.Y + 4, v.Y - half.Y - 4)
                target.Position = UDim2.new(cx / v.X, 0, cy / v.Y, 0)
            end
        end
    end)
    bind(UserInputService.InputEnded, function(input)
        if dragging and (input == startInput or input.UserInputType == Enum.UserInputType.MouseButton1) then
            dragging = false
            if moved then
                if opts.onMoved then opts.onMoved() end
            elseif opts.onClick then
                opts.onClick()
            end
        end
    end)
end
